name: Python Lint

on:
  push:
    branches:
      - main
    paths:
      - "samples/python/**"
      - ".github/workflows/python-lint.yaml"
  pull_request:
    types:
      - opened
      - reopened
      - synchronize
    paths:
      - "samples/python/**"
      - ".github/workflows/python-lint.yaml"
  schedule:
    - cron: "0 0 * * 0"

jobs:
  get_sample_dirs:
    runs-on: ubuntu-latest
    outputs:
      sample_dirs: ${{ steps.get_dirs.outputs.sample_dirs }}
    steps:
      - uses: actions/checkout@v4

      - name: Get Sample Directories
        id: get_dirs
        run: |
          # Build a JSON array of sample directories to feed the matrix.
          # Use Python to avoid depending on jq being installed on the runner.
          if [ ! -d "samples/python" ]; then
            # If samples/python does not exist, lint the repo root to avoid
            # creating an empty matrix which would cause the workflow to fail.
            echo "sample_dirs=[\".\"]" >> $GITHUB_OUTPUT
            echo "SAMPLE_DIRS: [\".\"]"
            exit 0
          fi

          SAMPLE_DIRS=$(python - <<'PY'
import os, json
base='samples/python'
out=[]
for root, dirs, files in os.walk(base):
    if root==base:
        continue
    # compute depth relative to base
    rel=root[len(base):]
    # normalize leading/trailing slashes
    rel = rel.strip(os.sep)
    depth = rel.count(os.sep) + (1 if rel else 0)
    if 1 <= depth <= 5:
        out.append(root)
# if no nested directories found, lint the base directory
if not out:
    out=[base]
print(json.dumps(out))
PY
)
          echo "sample_dirs=$SAMPLE_DIRS" >> $GITHUB_OUTPUT
          echo "SAMPLE_DIRS: $SAMPLE_DIRS"

  lint:
    runs-on: ubuntu-latest
    needs:
      - get_sample_dirs
    strategy:
      matrix:
        python-version: ["3.9", "3.10", "3.11", "3.12"]
        sample_dir: ${{ fromJson(needs.get_sample_dirs.outputs.sample_dirs) }}
      fail-fast: false

    steps:
      - uses: actions/checkout@v4

      - name: Set up Python ${{ matrix.python-version }}
        uses: actions/setup-python@v4
        with:
          python-version: ${{ matrix.python-version }}
          cache: "pip"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install "black>=23.0" "flake8>=6.0"

      - name: Run Black (check) and Flake8 in ${{ matrix.sample_dir }}
        working-directory: ${{ matrix.sample_dir }}
        run: |
          # Run black in check mode so CI fails when formatting is required.
          black --check --diff . || true
          # Run flake8 and fail the job on any issues.
          flake8 .
